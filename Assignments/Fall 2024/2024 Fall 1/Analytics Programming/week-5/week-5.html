#2) The lengths of pregnancies in a small rural village are normally distributed with a mean of 265 days and a standard deviation of 15 days.
#a. In what range would you expect to find the middle 50% of most pregnancies?
#b. What proportion of pregnancies will be more than 275 days?

# SOLUTION:

mean_pregnancy <- 265
sd_pregnancy <- 15

# a. Middle 50% (Q1 and Q3)
q1 <- qnorm(0.25, mean = mean_pregnancy, sd = sd_pregnancy)
q2 <- qnorm(0.75, mean=mean_pregnancy, sd=sd_pregnancy)
print(paste("Middle 50% range of pregnancies: ",q1 , q2))
## [1] "Middle 50% range of pregnancies:  254.882653747059 275.117346252941"
# b. Proportion of pregnancies > 275 days
prob_more_275 <- 1-pnorm(275,mean = mean_pregnancy,sd=sd_pregnancy)
print(paste("Proportion of pregnancies more than 275 days: ", prob_more_275))
## [1] "Proportion of pregnancies more than 275 days:  0.252492537546923"
#3) A certain car model has a mean gas mileage of 34 miles per gallon (mpg) with a standard deviation 5mpg. If a car is selected at random,

#a) What is the probability that the mileage of the selected car is greater than 33 mpg?
#b) What is the probability that the mileage of the car is between 33mpg and 35mpg?
  
#SOLUTION
  
mean_gas_mileage <- 34
mileage_sd <-5

#a selected car mileage is greater than 33 mpg
prob_mileage_greater_than_33 <- 1-pnorm(33,mean=mean_gas_mileage,sd=mileage_sd)
print(paste("Probability that the mileage os selected car is greater than 33 mpg: ", prob_mileage_greater_than_33))
## [1] "Probability that the mileage os selected car is greater than 33 mpg:  0.579259709439103"
#b mileage of the car is between 33mpg and 35 mpg
prob_mileage_btn_33_35 <- pnorm(35, mean=mean_gas_mileage, sd=mileage_sd) - pnorm(33, mean=mean_gas_mileage, sd=mileage_sd)
print(paste("Probability of mileage between 33 and 35 is: ", prob_mileage_btn_33_35))
## [1] "Probability of mileage between 33 and 35 is:  0.158519418878206"
